cmake_minimum_required(VERSION 3.16)
project(uwa_cell_tasks)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(ament_cmake_python REQUIRED)
find_package(gz_ros2_control REQUIRED)
find_package(hardware_interface REQUIRED)
find_package(pluginlib REQUIRED)
find_package(rclcpp REQUIRED)
find_package(std_srvs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(gz_sim_vendor REQUIRED)
find_package(gz-sim REQUIRED)
find_package(gz_plugin_vendor REQUIRED)
find_package(gz-plugin REQUIRED COMPONENTS register)
find_package(gz_transport_vendor REQUIRED)
find_package(gz-transport REQUIRED)
find_package(gz_msgs_vendor REQUIRED)
find_package(gz-msgs REQUIRED)
find_package(rosgraph_msgs REQUIRED)
find_package(nlohmann_json REQUIRED)
add_executable(task_clock_bridge src/task_clock_bridge.cpp)
target_link_libraries(task_clock_bridge ${rclcpp_TARGETS} ${rosgraph_msgs_TARGETS} gz-transport::gz-transport gz-msgs::gz-msgs)
add_library(contact_gripper SHARED src/contact_gripper.cpp src/task_telemetry.cpp)
add_library(force_probe SHARED src/force_probe.cpp)
target_link_libraries(force_probe PRIVATE nlohmann_json::nlohmann_json gz-sim::gz-sim gz-plugin::register)
add_library(force_telemetry SHARED src/force_telemetry.cpp)
target_link_libraries(force_telemetry PRIVATE ${rclcpp_TARGETS} ${std_msgs_TARGETS} nlohmann_json::nlohmann_json gz-sim::gz-sim gz-plugin::register)
target_link_libraries(contact_gripper ${gz_ros2_control_TARGETS} ${hardware_interface_TARGETS}
  ${pluginlib_TARGETS} ${rclcpp_TARGETS} ${std_srvs_TARGETS} ${std_msgs_TARGETS} ${rosgraph_msgs_TARGETS} nlohmann_json::nlohmann_json gz-sim::gz-sim gz-plugin::register)
pluginlib_export_plugin_description_file(gz_ros2_control hardware.xml)
ament_python_install_package(${PROJECT_NAME})
install(TARGETS contact_gripper force_probe force_telemetry LIBRARY DESTINATION lib)
install(TARGETS task_clock_bridge DESTINATION lib/${PROJECT_NAME})
install(PROGRAMS scripts/cell_task DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY config launch DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  add_executable(contact_transmission_test test/contact_transmission_test.cpp)
  target_include_directories(contact_transmission_test PRIVATE src)
  add_test(NAME contact_transmission COMMAND contact_transmission_test)
  find_package(ament_cmake_pytest REQUIRED)
  ament_add_pytest_test(task_core test/test_core.py TIMEOUT 90)
  ament_add_pytest_test(task_planning test/test_planning.py TIMEOUT 180)
  ament_add_pytest_test(task_guard test/test_guard.py TIMEOUT 90)
  ament_add_pytest_test(task_force_metrics test/test_force_metrics.py TIMEOUT 90)
  ament_add_pytest_test(task_operating test/test_operating.py TIMEOUT 90)
  ament_add_pytest_test(task_timing test/test_timing.py TIMEOUT 90)
  ament_add_pytest_test(task_exploration_driver test/test_exploration_driver.py TIMEOUT 90)
endif()
ament_package()
