cmake_minimum_required(VERSION 3.16)
project(uwa_cell_tasks)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(ament_cmake_python REQUIRED)
find_package(gz_ros2_control REQUIRED)
find_package(hardware_interface REQUIRED)
find_package(pluginlib REQUIRED)
find_package(rclcpp REQUIRED)
find_package(std_srvs REQUIRED)
find_package(gz_sim_vendor REQUIRED)
find_package(gz-sim REQUIRED)
add_library(contact_gripper SHARED src/contact_gripper.cpp)
target_link_libraries(contact_gripper ${gz_ros2_control_TARGETS} ${hardware_interface_TARGETS}
  ${pluginlib_TARGETS} ${rclcpp_TARGETS} ${std_srvs_TARGETS} gz-sim::gz-sim)
pluginlib_export_plugin_description_file(gz_ros2_control hardware.xml)
ament_python_install_package(${PROJECT_NAME})
install(TARGETS contact_gripper LIBRARY DESTINATION lib)
install(PROGRAMS scripts/cell_task DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY config launch DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(ament_cmake_pytest REQUIRED)
  ament_add_pytest_test(task_core test/test_core.py TIMEOUT 90)
  ament_add_pytest_test(task_planning test/test_planning.py TIMEOUT 180)
endif()
ament_package()
