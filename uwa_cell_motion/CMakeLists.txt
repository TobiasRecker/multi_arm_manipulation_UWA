cmake_minimum_required(VERSION 3.16)
project(uwa_cell_motion)
set(CMAKE_CXX_STANDARD 17)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
find_package(ament_cmake REQUIRED)
find_package(ament_cmake_python REQUIRED)
find_package(controller_interface REQUIRED)
find_package(hardware_interface REQUIRED)
find_package(pluginlib REQUIRED)
find_package(rclcpp REQUIRED)
find_package(moveit_core REQUIRED)
find_package(moveit_ros_planning REQUIRED)
find_package(ompl REQUIRED)
find_package(nlohmann_json REQUIRED)
find_package(urdf REQUIRED)
find_package(srdfdom REQUIRED)
add_library(collision_hold SHARED src/collision_hold.cpp)
target_link_libraries(collision_hold ${controller_interface_TARGETS} ${hardware_interface_TARGETS}
  ${pluginlib_TARGETS} ${rclcpp_TARGETS})
pluginlib_export_plugin_description_file(controller_interface controllers.xml)
add_executable(rrt_worker src/rrt_worker.cpp)
target_include_directories(rrt_worker PRIVATE ${OMPL_INCLUDE_DIRS})
target_link_libraries(rrt_worker ${moveit_core_TARGETS} ${moveit_ros_planning_TARGETS}
  ${rclcpp_TARGETS} ${urdf_TARGETS} ${srdfdom_TARGETS} ${OMPL_LIBRARIES} nlohmann_json::nlohmann_json)
ament_python_install_package(${PROJECT_NAME})
install(TARGETS collision_hold rrt_worker LIBRARY DESTINATION lib RUNTIME DESTINATION lib/${PROJECT_NAME})
install(PROGRAMS scripts/cell_motion DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY config launch DESTINATION share/${PROJECT_NAME})
ament_package()
